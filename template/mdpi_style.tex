%%
%% mdpi_style.tex  —  Academic document template (MDPI-inspired)
%%
%% Drop-in preamble used by generate.py.
%% No proprietary class files required; compiles with standard TeX Live / MiKTeX.
%%

\documentclass[11pt, a4paper, twoside]{article}

%% ── Encoding & Language ──────────────────────────────────────────────────────
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage[american]{babel}

%% ── Page Geometry (MDPI single-column) ───────────────────────────────────────
\usepackage[
  top=2.0cm, bottom=2.0cm,
  left=2.2cm, right=2.0cm,
  headheight=14pt
]{geometry}

%% ── Typography ───────────────────────────────────────────────────────────────
\usepackage{mathpazo}          % Palatino for body text
\usepackage[scaled=0.88]{helvet}
\usepackage{microtype}

%% ── Math ─────────────────────────────────────────────────────────────────────
\usepackage{amsmath, amssymb, bm}

%% ── Tables ───────────────────────────────────────────────────────────────────
\usepackage{booktabs, tabularx, multirow, array, makecell, colortbl}
\renewcommand{\arraystretch}{1.3}
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}}
\newcolumntype{C}[1]{>{\centering\arraybackslash}p{#1}}
\newcolumntype{R}[1]{>{\raggedleft\arraybackslash}p{#1}}

%% ── Figures ──────────────────────────────────────────────────────────────────
\usepackage{graphicx, float, caption}
\usepackage{subcaption}
\captionsetup{font=small, labelfont=bf, labelsep=period}

%% ── Colours ──────────────────────────────────────────────────────────────────
\usepackage{xcolor}
\definecolor{mdpiblue}{RGB}{0, 103, 172}
\definecolor{mcolor}{RGB}{26, 107, 58}
\definecolor{rowgray}{RGB}{242, 242, 242}
\definecolor{rowblue}{RGB}{217, 232, 247}

%% ── Section headings (MDPI style) ────────────────────────────────────────────
\usepackage{titlesec}
\titleformat{\section}
  {\normalfont\large\bfseries\color{mcolor}}
  {\thesection.}{0.5em}{}
\titleformat{\subsection}
  {\normalfont\normalsize\bfseries}
  {\thesubsection.}{0.5em}{}
\titleformat{\subsubsection}
  {\normalfont\normalsize\itshape}
  {\thesubsubsection.}{0.5em}{}
\titlespacing{\section}{0pt}{12pt}{4pt}
\titlespacing{\subsection}{0pt}{8pt}{3pt}
\titlespacing{\subsubsection}{0pt}{6pt}{2pt}

%% ── Headers & Footers ────────────────────────────────────────────────────────
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE,RO]{\small\thepage}
\fancyhead[LO]{\small\itshape\@shorttitle}
\fancyhead[RE]{\small\itshape\@shortauthor}
\renewcommand{\headrulewidth}{0.4pt}

%% ── Hyperlinks & References ──────────────────────────────────────────────────
\usepackage[
  colorlinks=true,
  linkcolor=mdpiblue,
  citecolor=mcolor,
  urlcolor=mdpiblue
]{hyperref}
\usepackage{natbib}
\setlength{\bibsep}{3pt}

%% ── Lists ────────────────────────────────────────────────────────────────────
\usepackage{enumitem}
\setlist{itemsep=2pt, topsep=3pt}

%% ── Line spacing ─────────────────────────────────────────────────────────────
\usepackage{setspace}
\setstretch{1.15}

%% ── Misc ─────────────────────────────────────────────────────────────────────
\usepackage{url, tcolorbox}
\tcbuselibrary{skins, breakable}
\setlength{\parskip}{4pt}
\setlength{\parindent}{0pt}
\setlength{\emergencystretch}{3em}
\hfuzz=6pt

%% ── MDPI-style abstract box ──────────────────────────────────────────────────
\newtcolorbox{abstractbox}{
  enhanced, breakable,
  colback=rowgray, colframe=mcolor,
  boxrule=0.6pt, arc=2pt,
  left=8pt, right=8pt, top=6pt, bottom=6pt,
  before upper={\setlength{\parskip}{4pt}},
  fonttitle=\bfseries\small\color{mcolor},
  title=Abstract
}

%% ── Coloured info boxes ──────────────────────────────────────────────────────
\newtcolorbox{notebox}[1][]{
  enhanced, colback=rowblue!40, colframe=mdpiblue,
  boxrule=0.5pt, arc=2pt,
  left=7pt, right=7pt, top=4pt, bottom=4pt,
  fonttitle=\bfseries\small, #1
}

%% ── Custom commands ──────────────────────────────────────────────────────────
\makeatletter
\newcommand{\shorttitle}[1]{\gdef\@shorttitle{#1}}
\newcommand{\shortauthor}[1]{\gdef\@shortauthor{#1}}
\newcommand{\journalname}[1]{\gdef\@journalname{#1}}
\shorttitle{}
\shortauthor{}
\journalname{}

%% ── Title block ──────────────────────────────────────────────────────────────
\renewcommand{\maketitle}{%
  \begin{flushleft}
    {\Large\bfseries\color{mcolor}\@title\par}
    \vspace{8pt}
    {\normalsize\@author\par}
    \vspace{4pt}
    {\small\itshape\@date\par}
    \vspace{2pt}
    \textcolor{mcolor}{\rule{\linewidth}{0.8pt}}
  \end{flushleft}
  \vspace{4pt}
}
\makeatother
